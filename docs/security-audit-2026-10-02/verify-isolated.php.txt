<?php
// Audit only. Run through Docker STDIN. All writes target connection-local TEMPORARY tables.
if (PHP_SAPI !== 'cli') {
    exit(1);
}
require '/var/www/html/glpi/vendor/autoload.php';
require GLPI_ROOT . '/config/config_db.php';
$DB = new DB();
$_SESSION = ['glpi_currenttime' => date('Y-m-d H:i:s')];
foreach (['config', 'login', 'loginip', 'lostpassword', 'whitelist'] as $class) {
    require GLPI_ROOT . '/plugins/moresecurity/inc/' . $class . '.class.php';
}

// Force synthetic names, protecting existing persistent tables even on an installed instance.
$classes = [PluginMoresecurityConfig::class, PluginMoresecurityLogin::class,
    PluginMoresecurityLoginip::class, PluginMoresecurityLostpassword::class,
    PluginMoresecurityWhitelist::class];
foreach ($classes as $i => $class) {
    $class::forceTable('glpi_audit_moresecurity_' . $i);
}
$definitions = [
    'id int PRIMARY KEY, number_attempts int DEFAULT 6, time_blocked int DEFAULT 300, max_time_blocked int DEFAULT 86400, attempts_reset int DEFAULT 3, time_reset int DEFAULT 300, ip_max_attempts int DEFAULT 5, ip_time_blocked int DEFAULT 300, ip_max_time_blocked int DEFAULT 86400, ip_threshold int DEFAULT 0, whitelisted_ips text NULL',
    'id int AUTO_INCREMENT PRIMARY KEY, login varchar(255), login_try int DEFAULT 0, blocked TIMESTAMP NULL, last_try TIMESTAMP NULL, UNIQUE KEY login(login)',
    'id int AUTO_INCREMENT PRIMARY KEY, login varchar(255), ip varchar(45), ip_try int DEFAULT 0, blocked TIMESTAMP NULL, last_try TIMESTAMP NULL, UNIQUE KEY login_ip(login,ip)',
    'id int AUTO_INCREMENT PRIMARY KEY, email varchar(255), email_try int DEFAULT 0, blocked TIMESTAMP NULL, KEY email(email)',
    'id int AUTO_INCREMENT PRIMARY KEY, name varchar(255), value varchar(255), type int DEFAULT 0, comment text NULL',
];
foreach ($classes as $i => $class) {
    $DB->doQuery('CREATE TEMPORARY TABLE ' . DBmysql::quoteName($class::getTable()) . ' (' . $definitions[$i] . ') ENGINE=InnoDB');
}
$DB->insert(PluginMoresecurityConfig::getTable(), ['id' => 1]);
$config = PluginMoresecurityConfig::getInstance();
$results = [];
function record(string $id, mixed $value): void {
    global $results;
    $results[$id] = $value;
}
function rows(string $class): array {
    global $DB;
    return array_values(iterator_to_array($DB->request(['FROM' => $class::getTable(), 'ORDER' => 'id'])));
}
function resetCounters(): void {
    global $DB;
    foreach ([PluginMoresecurityLogin::class, PluginMoresecurityLoginip::class, PluginMoresecurityLostpassword::class] as $class) {
        $DB->doQuery('DELETE FROM ' . DBmysql::quoteName($class::getTable()));
    }
}
record('runtime', ['php' => PHP_VERSION, 'php_tz' => date_default_timezone_get(), 'php_now' => date('Y-m-d H:i:s'),
    'sql_time' => $DB->fetchAssoc($DB->doQuery('SELECT NOW() AS db_now, @@session.time_zone AS db_tz')),
    'temporary_only' => true, 'native_db_abstraction' => get_class($DB)]);

// H02: actual GLPI criterion parser, actual plugin methods, two synthetic emails.
foreach (['one@audit.invalid', 'two@audit.invalid'] as $email) {
    $DB->insert(PluginMoresecurityLostpassword::getTable(), ['email' => $email]);
}
$input = ['LIKE', '%'];
$accepted = PluginMoresecurityLostpassword::checkEmail($input);
PluginMoresecurityLostpassword::addEmailTry($input);
record('email_array_mass_update', ['accepted' => $accepted, 'after' => rows(PluginMoresecurityLostpassword::class)]);
PluginMoresecurityLostpassword::addEmailTry($input);
PluginMoresecurityLostpassword::addEmailTry($input);
record('email_array_collective_lock', rows(PluginMoresecurityLostpassword::class));
resetCounters();
$DB->insert(PluginMoresecurityLostpassword::getTable(), ['email' => 'first@audit.invalid']);
$DB->insert(PluginMoresecurityLostpassword::getTable(), ['email' => 'blocked@audit.invalid', 'email_try' => 3, 'blocked' => date('Y-m-d H:i:s', time() + 300)]);
PluginMoresecurityLostpassword::checkEmail($input);
PluginMoresecurityLostpassword::addEmailTry($input);
record('email_array_clears_other_block', rows(PluginMoresecurityLostpassword::class));
resetCounters();

// Password-reset blocking alone never consults nor writes the IP budget.
$_SERVER['REMOTE_ADDR'] = '192.0.2.42';
for ($i = 0; $i < 3; $i++) {
    if (PluginMoresecurityLostpassword::checkEmail('victim@audit.invalid')) {
        PluginMoresecurityLostpassword::addEmailTry('victim@audit.invalid');
    }
}
record('reset_account_dos_without_ip_budget', ['fourth_allowed' => PluginMoresecurityLostpassword::checkEmail('victim@audit.invalid'),
    'emails' => rows(PluginMoresecurityLostpassword::class), 'ips' => rows(PluginMoresecurityLoginip::class)]);
resetCounters();

$DB->insert(PluginMoresecurityLostpassword::getTable(), ['email' => 'orphan@audit.invalid', 'email_try' => 3]);
record('reset_threshold_no_expiry', ['allowed' => PluginMoresecurityLostpassword::checkEmail('orphan@audit.invalid'),
    'after' => rows(PluginMoresecurityLostpassword::class)]);
resetCounters();

// Deterministic interleaving: all budget checks precede completions of in-flight attempts.
$config->fields['number_attempts'] = 3;
$config->fields['ip_max_attempts'] = 0;
$allowed = 0;
for ($i = 0; $i < 12; $i++) {
    $allowed += (int) PluginMoresecurityLogin::checkLogin('parallel@audit.invalid');
}
for ($i = 0; $i < $allowed; $i++) {
    PluginMoresecurityLogin::addLoginTry('parallel@audit.invalid');
}
record('login_budget_interleaving', ['threshold' => 3, 'admitted' => $allowed,
    'after' => rows(PluginMoresecurityLogin::class), 'note' => 'deterministic interleaving, not HTTP load']);
resetCounters();

// Regression: successful login clears only its own pair, preserving other accounts.
$config->fields['number_attempts'] = 6;
$config->fields['ip_max_attempts'] = 5;
foreach (['first@audit.invalid', 'second@audit.invalid'] as $login) {
    PluginMoresecurityLogin::checkLogin($login);
    PluginMoresecurityLogin::addLoginTry($login);
}
PluginMoresecurityLogin::clearLoginTry('first@audit.invalid');
record('success_keeps_other_ip_counters', rows(PluginMoresecurityLoginip::class));
resetCounters();

// Regression: expired block gets an expiry on reconciliation; backoff continues.
$DB->insert(PluginMoresecurityLogin::getTable(), ['login' => 'reconcile@audit.invalid', 'login_try' => 6]);
record('login_threshold_reconciled', ['allowed' => PluginMoresecurityLogin::checkLogin('reconcile@audit.invalid'),
    'after' => rows(PluginMoresecurityLogin::class)]);
resetCounters();
$past = date('Y-m-d H:i:s', time() - 10);
$DB->insert(PluginMoresecurityLogin::getTable(), ['login' => 'backoff@audit.invalid', 'login_try' => 6, 'last_try' => date('Y-m-d H:i:s'), 'blocked' => $past]);
$allowed = PluginMoresecurityLogin::checkLogin('backoff@audit.invalid');
PluginMoresecurityLogin::addLoginTry('backoff@audit.invalid');
record('backoff_after_expiry', ['allowed_after_expiry' => $allowed, 'after' => rows(PluginMoresecurityLogin::class)]);
resetCounters();

// Configuration disables account threshold but promises independent distributed-IP block.
$config->fields['number_attempts'] = 0;
$config->fields['ip_max_attempts'] = 0;
$config->fields['ip_threshold'] = 2;
foreach (['192.0.2.1', '198.51.100.1'] as $ip) {
    $_SERVER['REMOTE_ADDR'] = $ip;
    PluginMoresecurityLogin::checkLogin('distributed@audit.invalid');
    PluginMoresecurityLogin::addLoginTry('distributed@audit.invalid');
}
record('distributed_without_account_threshold', ['distinct_ips' => PluginMoresecurityLoginip::countDistinctIpsForLogin('distributed@audit.invalid'),
    'allowed' => PluginMoresecurityLogin::checkLogin('distributed@audit.invalid'), 'after' => rows(PluginMoresecurityLogin::class)]);

$config->fields['number_attempts'] = 2;
$_SERVER['REMOTE_ADDR'] = '203.0.113.1';
PluginMoresecurityLogin::addLoginTry('distributed@audit.invalid');
record('distributed_long_lock', rows(PluginMoresecurityLogin::class));
resetCounters();

// Zero durations are accepted by server-side clamping and yield ineffective blocking.
$config->fields['number_attempts'] = 1;
$config->fields['time_blocked'] = 0;
$config->fields['ip_max_attempts'] = 0;
$config->fields['ip_threshold'] = 0;
PluginMoresecurityLogin::checkLogin('zero@audit.invalid');
PluginMoresecurityLogin::addLoginTry('zero@audit.invalid');
record('zero_duration_login_allowed', ['allowed' => PluginMoresecurityLogin::checkLogin('zero@audit.invalid')]);
resetCounters();

// Scalar SQL quoting regression.
$payload = "audit' OR '1'='1";
PluginMoresecurityLogin::checkLogin($payload);
record('login_scalar_sql_literal', rows(PluginMoresecurityLogin::class));
resetCounters();

$normalize = new ReflectionMethod(PluginMoresecurityWhitelist::class, 'normalizeIpOrCidr');
$matches = new ReflectionMethod(PluginMoresecurityWhitelist::class, 'entryMatchesIp');
record('whitelist_regressions', [
    'ipv6_canonical' => $normalize->invoke(null, ' 2001:0db8:0:0:0:0:0:1 '),
    'bad_ipv4_rejected' => $normalize->invoke(null, '999.1.1.1') === null,
    'bad_cidr_rejected' => $normalize->invoke(null, '192.0.2.0/33') === null,
    'cidr_inside' => $matches->invoke(null, '192.0.2.0/24', inet_pton('192.0.2.42')),
    'cidr_outside' => $matches->invoke(null, '192.0.2.0/24', inet_pton('192.0.3.42')),
    'cross_family_rejected' => !$matches->invoke(null, '192.0.2.0/24', inet_pton('2001:db8::1')),
    'rightname' => PluginMoresecurityWhitelist::$rightname,
]);
$clamp = new ReflectionMethod(PluginMoresecurityConfig::class, 'clampInput');
record('config_clamping_regressions', [
    'negative' => $clamp->invoke(null, -5, 0, 100, 6),
    'huge' => $clamp->invoke(null, 99999, 0, 100, 6),
    'array_fallback' => $clamp->invoke(null, [], 0, 100, 6),
    'text_fallback' => $clamp->invoke(null, 'abc', 0, 100, 6),
    'duration_cap' => $clamp->invoke(null, 999999999999, 0, 31536000, 300),
    'zero_duration' => $clamp->invoke(null, 0, 0, 31536000, 300),
]);

// Runtime timing observation: SQL-side timestamps with PHP-side cutoffs.
$DB->doQuery('INSERT INTO ' . DBmysql::quoteName(PluginMoresecurityLoginip::getTable()) .
    " (login, ip, ip_try, last_try) VALUES ('eleven-hours', '192.0.2.99', 1, NOW() - INTERVAL 11 HOUR)");
$sum = new ReflectionMethod(PluginMoresecurityLoginip::class, 'sumTriesForIp');
record('mixed_time_window_observation', ['sql_age_hours' => 11, 'configured_window_hours' => 12,
    'counted_attempts' => $sum->invoke(null, '192.0.2.99')]);

$assertions = [
    'array_updates_all_emails' => array_column($results['email_array_mass_update']['after'], 'email_try') === [1, 1],
    'array_blocks_all_emails' => count(array_filter($results['email_array_collective_lock'], fn($r) => $r['blocked'] !== null)) === 2,
    'array_clears_other_block' => $results['email_array_clears_other_block'][1]['blocked'] === null,
    'reset_ignores_ip_budget' => $results['reset_account_dos_without_ip_budget']['ips'] === [],
    'reset_orphan_never_reconciled' => !$results['reset_threshold_no_expiry']['allowed'] && $results['reset_threshold_no_expiry']['after'][0]['blocked'] === null,
    'inflight_checks_exceed_budget' => $results['login_budget_interleaving']['admitted'] === 12,
    'other_account_ip_preserved' => $results['success_keeps_other_ip_counters'][1]['ip_try'] === 1,
    'login_orphan_reconciled' => $results['login_threshold_reconciled']['after'][0]['blocked'] !== null,
    'backoff_increments_after_expiry' => $results['backoff_after_expiry']['after'][0]['login_try'] === 7,
    'distributed_inactive_without_account_limit' => $results['distributed_without_account_threshold']['allowed'] === true,
    'distributed_lock_until_2037' => $results['distributed_long_lock'][0]['blocked'] === '2037-12-31 23:59:59',
    'zero_duration_ineffective' => $results['zero_duration_login_allowed']['allowed'] === true,
    'scalar_sql_payload_stays_literal' => $results['login_scalar_sql_literal'][0]['login'] === $payload,
    'whitelist_requires_config' => $results['whitelist_regressions']['rightname'] === 'config',
    'invalid_cidr_rejected' => $results['whitelist_regressions']['bad_cidr_rejected'],
    'cidr_inside_matches' => $results['whitelist_regressions']['cidr_inside'],
    'cidr_outside_does_not_match' => !$results['whitelist_regressions']['cidr_outside'],
    'config_array_fallback' => $results['config_clamping_regressions']['array_fallback'] === 6,
];
record('assertions', $assertions);
echo json_encode($results, JSON_PRETTY_PRINT | JSON_UNESCAPED_SLASHES) . "\n";
exit(in_array(false, $assertions, true) ? 1 : 0);
// Temporary tables are automatically removed when this process closes its connection.
