$fail = 0; $pass = 0;
function ok(string $name, bool $cond, string $extra = ''): void {
    global $fail, $pass;
    echo ($cond ? "  PASS " : "  FAIL ") . $name . ($cond || !$extra ? '' : "  -> $extra") . "\n";
    $cond ? $pass++ : $fail++;
}
function parallel(array $cfgv, string $kind, array $pairs): array {
    $procs = [];
    foreach ($pairs as [$a, $b]) {
        $cmd = ['php', __FILE__, 'worker', json_encode($cfgv), $kind, $a, $b];
        $p = proc_open($cmd, [1 => ['pipe', 'w'], 2 => ['pipe', 'w']], $pipes);
        $procs[] = [$p, $pipes];
    }
    $out = [];
    foreach ($procs as [$p, $pipes]) {
        $out[] = trim(stream_get_contents($pipes[1]));
        fclose($pipes[1]); fclose($pipes[2]); proc_close($p);
    }
    return $out;
}
